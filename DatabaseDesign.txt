-- =========================================================
-- CREDIT CARD MANAGEMENT SYSTEM
-- Oracle Database
-- =========================================================

-- ---------------------------------------------------------
-- 1. CUSTOMER
-- ---------------------------------------------------------
CREATE TABLE CUSTOMER (
    CUSTOMER_ID      NUMBER GENERATED BY DEFAULT AS IDENTITY,
    CUSTOMER_NAME    VARCHAR2(100) NOT NULL,
    EMAIL_ADDRESS    VARCHAR2(150) NOT NULL,
    MOBILE_NUMBER    VARCHAR2(20) NOT NULL,
    PAN_NUMBER       VARCHAR2(20) NOT NULL,

    CONSTRAINT PK_CUSTOMER
        PRIMARY KEY (CUSTOMER_ID),

    CONSTRAINT UK_CUSTOMER_EMAIL
        UNIQUE (EMAIL_ADDRESS),

    CONSTRAINT UK_CUSTOMER_MOBILE
        UNIQUE (MOBILE_NUMBER),

    CONSTRAINT UK_CUSTOMER_PAN
        UNIQUE (PAN_NUMBER)
);


-- ---------------------------------------------------------
-- 2. CREDIT_CARD
-- ---------------------------------------------------------
CREATE TABLE CREDIT_CARD (
    CARD_NUMBER          VARCHAR2(20),
    CUSTOMER_ID          NUMBER NOT NULL,
    CARD_TYPE            VARCHAR2(20) NOT NULL,
    CREDIT_LIMIT         NUMBER(15,2) NOT NULL,
    AVAILABLE_CREDIT     NUMBER(15,2) NOT NULL,
    OUTSTANDING_AMOUNT   NUMBER(15,2) NOT NULL,
    EXPIRY_DATE          DATE NOT NULL,
    CARD_STATUS          VARCHAR2(20) NOT NULL,

    CONSTRAINT PK_CREDIT_CARD
        PRIMARY KEY (CARD_NUMBER),

    CONSTRAINT FK_CARD_CUSTOMER
        FOREIGN KEY (CUSTOMER_ID)
        REFERENCES CUSTOMER(CUSTOMER_ID),

    CONSTRAINT CK_CARD_TYPE
        CHECK (CARD_TYPE IN ('SILVER', 'GOLD', 'PLATINUM')),

    CONSTRAINT CK_CARD_STATUS
        CHECK (CARD_STATUS IN ('ACTIVE', 'BLOCKED')),

    CONSTRAINT CK_CARD_LIMITS
        CHECK (
            CREDIT_LIMIT >= 0
            AND AVAILABLE_CREDIT >= 0
            AND OUTSTANDING_AMOUNT >= 0
        ),

    CONSTRAINT CK_CARD_CREDIT_BALANCE
        CHECK (AVAILABLE_CREDIT + OUTSTANDING_AMOUNT <= CREDIT_LIMIT)
);


-- ---------------------------------------------------------
-- 3. MERCHANT
-- ---------------------------------------------------------
CREATE TABLE MERCHANT (
    MERCHANT_ID       NUMBER GENERATED BY DEFAULT AS IDENTITY,
    MERCHANT_NAME     VARCHAR2(150) NOT NULL,
    CATEGORY          VARCHAR2(100) NOT NULL,
    LOCATION          VARCHAR2(200) NOT NULL,

    CONSTRAINT PK_MERCHANT
        PRIMARY KEY (MERCHANT_ID)
);


-- ---------------------------------------------------------
-- 4. BILLS
-- ---------------------------------------------------------
CREATE TABLE BILLS (
    BILL_ID                NUMBER GENERATED BY DEFAULT AS IDENTITY,
    CARD_NUMBER            VARCHAR2(20) NOT NULL,
    BILL_AMOUNT            NUMBER(15,2) NOT NULL,
    PAID_AMOUNT            NUMBER(15,2) DEFAULT 0 NOT NULL,
    OUTSTANDING_AMOUNT     NUMBER(15,2) NOT NULL,

    CONSTRAINT PK_BILLS
        PRIMARY KEY (BILL_ID),

    CONSTRAINT FK_BILL_CARD
        FOREIGN KEY (CARD_NUMBER)
        REFERENCES CREDIT_CARD(CARD_NUMBER),

    CONSTRAINT CK_BILL_AMOUNTS
        CHECK (
            BILL_AMOUNT >= 0
            AND PAID_AMOUNT >= 0
            AND OUTSTANDING_AMOUNT >= 0
            AND PAID_AMOUNT <= BILL_AMOUNT
        )
);


-- ---------------------------------------------------------
-- 5. TRANSACTION
-- ---------------------------------------------------------
CREATE TABLE TRANSACTION (
    TRANSACTION_ID       NUMBER GENERATED BY DEFAULT AS IDENTITY,
    CARD_NUMBER          VARCHAR2(20) NOT NULL,
    TRANSACTION_TYPE     VARCHAR2(20) NOT NULL,
    AMOUNT               NUMBER(15,2) NOT NULL,
    MERCHANT_ID          NUMBER,
    TRANSACTION_DATE     TIMESTAMP DEFAULT CURRENT_TIMESTAMP NOT NULL,
    STATUS                VARCHAR2(20) NOT NULL,

    CONSTRAINT PK_TRANSACTION
        PRIMARY KEY (TRANSACTION_ID),

    CONSTRAINT FK_TRANSACTION_CARD
        FOREIGN KEY (CARD_NUMBER)
        REFERENCES CREDIT_CARD(CARD_NUMBER),

    CONSTRAINT FK_TRANSACTION_MERCHANT
        FOREIGN KEY (MERCHANT_ID)
        REFERENCES MERCHANT(MERCHANT_ID),

    CONSTRAINT CK_TRANSACTION_TYPE
        CHECK (TRANSACTION_TYPE IN ('PURCHASE', 'PAYMENT')),

    CONSTRAINT CK_TRANSACTION_STATUS
        CHECK (STATUS IN ('SUCCESS', 'FAILED')),

    CONSTRAINT CK_TRANSACTION_AMOUNT
        CHECK (AMOUNT > 0)
);
